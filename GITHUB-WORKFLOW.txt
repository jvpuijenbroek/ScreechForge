name: Build Windows VST3
on:
  workflow_dispatch:
  push:
    branches: [main, master]
permissions:
  contents: read
jobs:
  build:
    runs-on: windows-2022
    timeout-minutes: 40
    steps:
      - uses: actions/checkout@v4
      - name: Configure
        run: cmake -S . -B build -G "Visual Studio 17 2022" -A x64
      - name: Build instrument
        run: cmake --build build --config Release --target ScreechForge_VST3 --parallel 2
      - name: Verify and stage complete VST3 bundle
        shell: pwsh
        run: |
          $bundle = 'build/ScreechForge_artefacts/Release/VST3/ScreechForge.vst3'
          if (!(Test-Path "$bundle/Contents/x86_64-win/ScreechForge.vst3")) {
            throw 'Expected VST3 binary missing'
          }
          New-Item -ItemType Directory -Path dist -Force | Out-Null
          Copy-Item $bundle dist -Recurse
      - name: Upload Windows VST3
        uses: actions/upload-artifact@v7
        with:
          name: ScreechForge-Windows-VST3
          path: dist/
          if-no-files-found: error
          retention-days: 14
